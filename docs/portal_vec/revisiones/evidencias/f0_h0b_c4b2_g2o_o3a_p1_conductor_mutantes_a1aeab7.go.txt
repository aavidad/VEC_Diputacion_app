func autoprobarMutantes(directorio, temporal string) error {
	for _, nombre := range []string{"M01", "M02", "M03", "M04", "M05", "M06", "M07", "M08", "M10", "M11"} {
		if err := exigirPlazoGlobal(); err != nil {
			return err
		}
		datos, err := leerMutante(directorio, nombre)
		if err != nil {
			return fmt.Errorf("%s no acreditado: %w", nombre, err)
		}
		runner, err := crearRunnerPrivado(datos, temporal)
		if err != nil {
			return fmt.Errorf("%s sin runner privado: %w", nombre, err)
		}
		err = ejecutarMutante(nombre, runner, temporal)
		err = errors.Join(err, runner.archivo.Close())
		if err != nil {
			return fmt.Errorf("%s sobrevivió o no limpió: %w", nombre, err)
		}
	}
	if err := exigirPlazoGlobal(); err != nil {
		return err
	}
	datos, err := leerMutante(directorio, "M09")
	if err != nil {
		return err
	}
	if strings.Count(string(datos), "/usr/bin/env -0 |") != 2 {
		return errors.New("M09 no conserva dos pipelines")
	}
	return nil
}

func leerMutante(directorio, nombre string) ([]byte, error) {
	esperado, existe := metadatosMutantes[nombre]
	if !existe {
		return nil, errors.New("mutante no enumerado")
	}
	ruta := filepath.Join(directorio, nombre+".sh")
	fd, err := syscall.Open(ruta, syscall.O_RDONLY|syscall.O_NONBLOCK|syscall.O_CLOEXEC|syscall.O_NOFOLLOW, 0)
	if err != nil {
		return nil, err
	}
	archivo := os.NewFile(uintptr(fd), "mutante")
	defer archivo.Close()
	info, err := archivo.Stat()
	if err != nil || !info.Mode().IsRegular() || info.Mode().Perm()&0002 != 0 || info.Size() != esperado.bytes {
		return nil, fmt.Errorf("forma mutante inválida: modo=%v err=%v", info, err)
	}
	datos, err := leerArchivoAcotado(archivo, esperado.bytes)
	if err != nil {
		return nil, err
	}
	if obtenida := fmt.Sprintf("%x", sha256.Sum256(datos)); obtenida != esperado.huella {
		return nil, fmt.Errorf("huella=%s", obtenida)
	}
	return datos, nil
}

func ejecutarMutante(nombre string, runner *runnerPrivado, temporal string) error {
	switch nombre {
	case "M01":
		return mutanteBloqueado(runner, temporal, nombre, "/usr/bin/env -0")
	case "M03":
		return mutanteBloqueado(runner, temporal, nombre, "/usr/bin/true")
	case "M02":
		return mutanteEntornoAntesSTOP(runner, temporal, nombre)
	case "M04":
		return mutanteOrdenAntesSTOP(runner, temporal, nombre)
	case "M05":
		return mutanteFD9Abierto(runner, temporal, nombre)
	case "M06":
		return mutanteSegundaAceptada(runner, temporal, nombre)
	case "M07":
		return mutanteAdversoDetenido(runner, temporal, nombre)
	case "M08", "M10":
		return mutanteEntornoOmitido(runner, temporal, nombre)
	case "M11":
		return mutanteRaizAntesEntorno(runner, temporal, nombre)
	default:
		return errors.New("mutante conductual no enumerado")
	}
}

func mutanteBloqueado(runner *runnerPrivado, temporal, nombre, orden string) (err error) {
	e, err := iniciar(runner, temporal, nombre, true, "")
	if err != nil {
		return err
	}
	defer func() {
		if !e.esperado {
			err = errors.Join(err, e.limpiar(limiteLimpieza()))
		}
	}()
	limite := limiteCaso()
	traza, err := esperarAnclaTraza(e.traza, "read -r", limite)
	if err != nil {
		return err
	}
	if _, err = esperarEstado(e, 'S', limite); err != nil {
		return err
	}
	indiceOrden := strings.Index(traza, orden)
	indiceRead := strings.LastIndex(traza, "read -r")
	if err != nil || indiceOrden < 0 || indiceRead <= indiceOrden || contieneSTOP(traza) {
		return fmt.Errorf("orden externo no completado antes del read: orden=%d read=%d stop=%v err=%v", indiceOrden, indiceRead, contieneSTOP(traza), err)
	}
	if err = e.limpiar(limiteLimpieza()); err != nil {
		return err
	}
	return e.validarYRetirarEvidencia()
}

func mutanteEntornoAntesSTOP(runner *runnerPrivado, temporal, nombre string) (err error) {
	e, err := iniciar(runner, temporal, nombre, true, "LD")
	if err != nil {
		return err
	}
	defer func() {
		if !e.esperado {
			err = errors.Join(err, e.limpiar(limiteLimpieza()))
		}
	}()
	if err = e.escribirYCerrar(ticketValido(os.Getpid())); err != nil {
		return err
	}
	if err = e.recoger(65, limiteCaso()); err != nil {
		return err
	}
	traza, err := leerTraza(e.traza)
	if err != nil || !contieneEntorno(traza) || contieneSTOP(traza) {
		return errors.New("M02 no ejecutó entorno antes de STOP")
	}
	return e.validarYRetirarEvidencia()
}

func mutanteEntornoOmitido(runner *runnerPrivado, temporal, nombre string) (err error) {
	e, err := iniciar(runner, temporal, nombre, false, "LD")
	if err != nil {
		return err
	}
	defer func() {
		if !e.esperado {
			err = errors.Join(err, e.limpiar(limiteLimpieza()))
		}
	}()
	if err = e.cerrarEscritor(); err != nil {
		return err
	}
	if err = e.recoger(65, limiteCaso()); err != nil {
		return err
	}
	traza, err := leerTraza(e.traza)
	dirname := strings.Index(traza, "dirname --")
	entorno := strings.Index(traza, "/usr/bin/env -0")
	if err != nil || dirname < 0 || contieneSTOP(traza) ||
		(nombre == "M08" && entorno >= 0) || (nombre == "M10" && (entorno < 0 || dirname <= entorno)) {
		return errors.New("M08/M10 aceptó LD sin alcanzar la raíz")
	}
	return validarFalloRaizYRetirar(e)
}

func mutanteRaizAntesEntorno(runner *runnerPrivado, temporal, nombre string) (err error) {
	e, err := iniciar(runner, temporal, nombre, false, "LD")
	if err != nil {
		return err
	}
	defer func() {
		if !e.esperado {
			err = errors.Join(err, e.limpiar(limiteLimpieza()))
		}
	}()
	if err = e.cerrarEscritor(); err != nil {
		return err
	}
	if err = e.recoger(65, limiteCaso()); err != nil {
		return err
	}
	traza, err := leerTraza(e.traza)
	dirname := strings.Index(traza, "dirname --")
	entorno := strings.Index(traza, "/usr/bin/env -0")
	if err != nil || dirname < 0 || (entorno >= 0 && entorno <= dirname) || contieneSTOP(traza) {
		return errors.New("M11 no resolvió raíz antes del entorno")
	}
	if entorno < 0 {
		return validarFalloRaizYRetirar(e)
	}
	return e.validarYRetirarEvidencia()
}

func contieneSTOP(traza string) bool {
	return strings.Contains(traza, "builtin kill -STOP")
}

func validarFalloRaizYRetirar(e *ejecucion) (resultado error) {
	salida, err := leerRutaAcotada(e.salida, 65536)
	if err != nil || len(salida) != 0 {
		resultado = errors.Join(resultado, errors.New("stdout del fallo de raíz no vacío"), err)
	}
	errorSalida, err := leerRutaAcotada(e.error, 65536)
	patron := regexp.MustCompile(`^/proc/self/fd/8: line [0-9]+: cd: /proc/[0-9]+/fd/\.\./\.\./\.\.: No such file or directory\n$`)
	if err != nil || !patron.Match(errorSalida) {
		resultado = errors.Join(resultado, errors.New("stderr del fallo de raíz discrepante"), err)
	}
	for _, ruta := range []string{e.traza, e.salida, e.error} {
		resultado = errors.Join(resultado, os.Remove(ruta))
	}
	return resultado
}

func esperarAnclaTraza(ruta, ancla string, limite time.Time) (string, error) {
	for time.Now().Before(limite) {
		traza, err := leerTraza(ruta)
		if err == nil && strings.Contains(traza, ancla) {
			return traza, nil
		}
		if err != nil {
			return "", err
		}
		time.Sleep(time.Millisecond)
	}
	return "", errors.New("ancla xtrace no observada")
}
