type escritorAcotado struct {
	buffer bytes.Buffer
	maximo int
}

var (
	raizSalidas    *os.Root
	salidasCreadas []string
	salidasValidas bool
)

func (e *escritorAcotado) Write(p []byte) (int, error) {
	if len(p) > e.maximo-e.buffer.Len() {
		return 0, errors.New("salida excede el límite")
	}
	return e.buffer.Write(p)
}

func git(raiz string, argumentos ...string) (string, error) {
	info, err := os.Lstat("/usr/bin/git")
	if err != nil || !info.Mode().IsRegular() || info.Mode().Perm()&0022 != 0 || info.Size() <= 0 || info.Size() > 16<<20 {
		return "", errors.New("binario Git no acreditado")
	}
	forma, ok := info.Sys().(*syscall.Stat_t)
	if !ok || forma.Uid != 0 || forma.Nlink == 0 {
		return "", errors.New("propiedad de Git no acreditada")
	}
	ctx, cancelar := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancelar()
	args := []string{
		"--no-pager", "--no-optional-locks", "--no-replace-objects", "--literal-pathspecs",
		"-c", "core.fsmonitor=false",
		"-c", "core.hooksPath=/dev/null",
		"-c", "core.pager=cat",
		"-c", "color.ui=false",
		"-C", raiz,
	}
	args = append(args, argumentos...)
	cmd := exec.CommandContext(ctx, "/usr/bin/git", args...)
	cmd.Dir = "/"
	// No se hereda ninguna variable GIT_*, de localización, cargador o usuario.
	cmd.Env = []string{
		"GIT_CONFIG_GLOBAL=/dev/null",
		"GIT_CONFIG_NOSYSTEM=1",
		"GIT_CONFIG_SYSTEM=/dev/null",
		"GIT_NO_REPLACE_OBJECTS=1",
		"GIT_PAGER=cat",
		"GIT_TERMINAL_PROMPT=0",
		"HOME=/nonexistent",
		"LANG=C",
		"LC_ALL=C",
		"PATH=/usr/bin:/bin",
	}
	salida := &escritorAcotado{maximo: 65536}
	diagnostico := &escritorAcotado{maximo: 65536}
	cmd.Stdout, cmd.Stderr = salida, diagnostico
	err = cmd.Run()
	if ctx.Err() != nil {
		return "", errors.New("git excedió el plazo")
	}
	if err != nil {
		return "", errors.New("git rechazó la comprobación")
	}
	return salida.buffer.String(), nil
}

func exigir(condicion bool, mensaje string) {
	if !condicion {
		retirarSalidas()
		fmt.Fprintln(os.Stderr, mensaje)
		os.Exit(65)
	}
}

func retirarSalidas() {
	if raizSalidas == nil || salidasValidas {
		return
	}
	for i := len(salidasCreadas) - 1; i >= 0; i-- {
		_ = raizSalidas.Remove(salidasCreadas[i])
	}
}

func acreditarGenealogia(raiz string) string {
	absoluta, err := filepath.Abs(raiz)
	exigir(err == nil, "raíz no acreditada")
	absoluta = filepath.Clean(absoluta)
	real, err := filepath.EvalSymlinks(absoluta)
	exigir(err == nil && real == absoluta, "raíz no canónica")
	top, err := git(real, "rev-parse", "--show-toplevel")
	exigir(err == nil, "repositorio no acreditado")
	topReal, err := filepath.EvalSymlinks(strings.TrimSpace(top))
	exigir(err == nil && topReal == real, "repositorio no acreditado")
	head, err := git(real, "rev-parse", "HEAD")
	exigir(err == nil, "HEAD no acreditado")
	head = strings.TrimSpace(head)
	if head == shaMaterial {
		padre, err := git(real, "rev-parse", "HEAD^")
		exigir(err == nil && strings.TrimSpace(padre) == shaPadreMaterial, "padre material discrepante")
		diff, err := git(real, "diff", "--name-only", "--")
		exigir(err == nil && diff == "", "cambios seguidos no autorizados")
		indice, err := git(real, "diff", "--cached", "--name-only", "--")
		exigir(err == nil && indice == "", "índice no autorizado")
		argumentos := []string{"status", "--porcelain", "--untracked-files=all", "--"}
		argumentos = append(argumentos, rutasEvidencia...)
		estado, err := git(real, argumentos...)
		exigir(err == nil, "estado precommit no acreditado")
		lineas := strings.Split(strings.TrimSpace(estado), "\n")
		esperadas := make([]string, len(rutasEvidencia))
		for i, ruta := range rutasEvidencia {
			esperadas[i] = "?? " + ruta
		}
		sort.Strings(lineas)
		sort.Strings(esperadas)
		exigir(strings.Join(lineas, "\n") == strings.Join(esperadas, "\n"), "evidencia precommit incompleta")
	} else {
		padre, err := git(real, "rev-parse", "HEAD^")
		exigir(err == nil && strings.TrimSpace(padre) == shaMaterial, "commit de evidencia sin padre material")
		abuelo, err := git(real, "rev-parse", "HEAD^^")
		exigir(err == nil && strings.TrimSpace(abuelo) == shaPadreMaterial, "genealogía material discrepante")
		estado, err := git(real, "status", "--porcelain", "--untracked-files=all")
		exigir(err == nil && estado == "", "postcommit no limpio")
		cambios, err := git(real, "diff-tree", "--no-commit-id", "--name-status", "-r", "HEAD")
		exigir(err == nil, "commit de evidencia no acreditado")
		lineas := strings.Split(strings.TrimSpace(cambios), "\n")
		esperadas := make([]string, len(rutasEvidencia))
		for i, ruta := range rutasEvidencia {
			esperadas[i] = "A\t" + ruta
		}
		sort.Strings(lineas)
		sort.Strings(esperadas)
		exigir(strings.Join(lineas, "\n") == strings.Join(esperadas, "\n"), "write-set del commit de evidencia discrepante")
		for _, ruta := range rutasEvidencia {
			arbol, err := git(real, "ls-tree", "HEAD", "--", ruta)
			exigir(err == nil && strings.HasPrefix(arbol, "100644 blob "), "modo Git de evidencia discrepante")
		}
	}
	for _, ruta := range rutasEvidencia {
		info, err := os.Lstat(filepath.Join(real, filepath.FromSlash(ruta)))
		exigir(err == nil && info.Mode().IsRegular() && info.Mode().Perm() == 0644, "forma física de evidencia discrepante")
	}
	return real
}

func leerCandidato(ruta string) []byte {
	fd, err := syscall.Open(ruta, syscall.O_RDONLY|syscall.O_NONBLOCK|syscall.O_CLOEXEC|syscall.O_NOFOLLOW, 0)
	exigir(err == nil, "runner no accesible")
	archivo := os.NewFile(uintptr(fd), "runner-material")
	defer archivo.Close()
	info, err := archivo.Stat()
	exigir(err == nil && info.Mode().IsRegular() && info.Size() == tamanoRunner, "forma del runner discrepante")
	permisos := info.Mode().Perm()
	exigir(permisos == 0755 || permisos == 0775, "modo físico del runner discrepante")
	datos, err := io.ReadAll(io.LimitReader(archivo, tamanoRunner+1))
	exigir(err == nil && int64(len(datos)) == tamanoRunner, "lectura del runner discrepante")
	exigir(fmt.Sprintf("%x", sha256.Sum256(datos)) == shaCandidato, "huella del runner discrepante")
	return datos
}

func leerFuenteConductor(raiz string, fuente fuenteConductor) []byte {
	ruta := filepath.Join(raiz, filepath.FromSlash(fuente.ruta))
	fd, err := syscall.Open(ruta, syscall.O_RDONLY|syscall.O_NONBLOCK|syscall.O_CLOEXEC|syscall.O_NOFOLLOW, 0)
	exigir(err == nil, "fuente del conductor no accesible")
	archivo := os.NewFile(uintptr(fd), "fuente-conductor")
	defer archivo.Close()
	info, err := archivo.Stat()
	exigir(err == nil && info.Mode().IsRegular() && info.Mode().Perm() == 0644 && info.Size() == fuente.tamano, "forma de fuente del conductor discrepante")
	datos, err := io.ReadAll(io.LimitReader(archivo, fuente.tamano+1))
	exigir(err == nil && int64(len(datos)) == fuente.tamano, "lectura de fuente del conductor discrepante")
	exigir(fmt.Sprintf("%x", sha256.Sum256(datos)) == fuente.huella, "huella de fuente del conductor discrepante")
	return datos
}

func leerMaterialLigado(raiz string) []byte {
	datos := leerCandidato(filepath.Join(raiz, filepath.FromSlash(rutaRunner)))
	arbol, err := git(raiz, "ls-tree", "HEAD", "--", rutaRunner)
	exigir(err == nil, "entrada Git del runner ausente")
	campos := strings.Fields(arbol)
	exigir(len(campos) == 4 && campos[0] == "100755" && campos[1] == "blob" && len(campos[2]) == 40 && campos[3] == rutaRunner, "entrada Git del runner discrepante")
	blob, err := git(raiz, "cat-file", "blob", "HEAD:"+rutaRunner)
	exigir(err == nil && len(blob) == int(tamanoRunner) && bytes.Equal(datos, []byte(blob)), "runner físico no ligado al blob de HEAD")
	return datos
}

func una(datos []byte, viejo, nuevo string) []byte {
	exigir(bytes.Count(datos, []byte(viejo)) == 1, "patrón mutante no único")
	return bytes.Replace(datos, []byte(viejo), []byte(nuevo), 1)
}

func mover(datos []byte, ancla string) []byte {
	datos = una(datos, bloque, "")
	return una(datos, ancla, bloque+ancla)
}

func baseDesdeCandidato(datos []byte) []byte {
	base := una(datos, bloque, "")
	base = una(base, "export LC_ALL=C\n", bloque+"export LC_ALL=C\n")
	exigir(fmt.Sprintf("%x", sha256.Sum256(base)) == shaBase, "reversión base discrepante")
	return base
}

func escribir(raiz *os.Root, nombre string, datos []byte) {
	archivo, err := raiz.OpenFile(nombre, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	exigir(err == nil, "no se pudo crear una salida")
	salidasCreadas = append(salidasCreadas, nombre)
	n, err := archivo.Write(datos)
	errCierre := archivo.Close()
	exigir(err == nil && n == len(datos) && errCierre == nil, "escritura de salida incompleta")
}
