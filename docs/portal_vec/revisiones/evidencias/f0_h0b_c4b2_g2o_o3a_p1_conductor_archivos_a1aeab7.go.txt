package main

import (
	"crypto/sha256"
	"errors"
	"fmt"
	"io"
	"os"
	"syscall"
)

func leerRunnerAcreditado(ruta string) ([]byte, error) {
	fd, err := syscall.Open(ruta, syscall.O_RDONLY|syscall.O_NONBLOCK|syscall.O_CLOEXEC|syscall.O_NOFOLLOW, 0)
	if err != nil {
		return nil, err
	}
	archivo := os.NewFile(uintptr(fd), "runner-entrada")
	defer archivo.Close()
	info, err := archivo.Stat()
	if err != nil {
		return nil, err
	}
	permisos := info.Mode().Perm()
	if info.Size() != 46123 || !info.Mode().IsRegular() ||
		(permisos != 0755 && permisos != 0775) {
		return nil, fmt.Errorf("runner no acreditado: bytes=%d modo=%o", info.Size(), permisos)
	}
	datos, err := leerArchivoAcotado(archivo, 46123)
	if err != nil {
		return nil, err
	}
	if huella := fmt.Sprintf("%x", sha256.Sum256(datos)); huella != huellaRunner {
		return nil, fmt.Errorf("runner no acreditado: sha=%s", huella)
	}
	return datos, nil
}

func crearRunnerPrivado(datos []byte, temporal string) (*runnerPrivado, error) {
	ruta := temporal + "/runner-privado"
	escritor, err := os.OpenFile(ruta, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if err != nil {
		return nil, err
	}
	if n, err := escritor.Write(datos); err != nil || n != len(datos) {
		_ = escritor.Close()
		return nil, fmt.Errorf("escritura privada: n=%d err=%v", n, err)
	}
	if err = escritor.Sync(); err != nil {
		_ = escritor.Close()
		return nil, err
	}
	if err = escritor.Close(); err != nil {
		return nil, err
	}
	fd, err := syscall.Open(ruta, syscall.O_RDONLY|syscall.O_CLOEXEC|syscall.O_NOFOLLOW, 0)
	if err != nil {
		return nil, err
	}
	archivo := os.NewFile(uintptr(fd), "runner-privado")
	if err = os.Remove(ruta); err != nil {
		_ = archivo.Close()
		return nil, err
	}
	var forma syscall.Stat_t
	if err = syscall.Fstat(fd, &forma); err != nil || forma.Nlink != 0 || forma.Mode&0777 != 0600 || forma.Size != int64(len(datos)) {
		_ = archivo.Close()
		return nil, fmt.Errorf("forma privada: err=%v enlaces=%d modo=%o bytes=%d", err, forma.Nlink, forma.Mode&0777, forma.Size)
	}
	return &runnerPrivado{archivo: archivo, dev: uint64(forma.Dev), ino: forma.Ino, tamano: forma.Size}, nil
}

func (r *runnerPrivado) duplicar() (*os.File, error) {
	descriptor, _, errno := syscall.Syscall(syscall.SYS_FCNTL, r.archivo.Fd(), uintptr(syscall.F_DUPFD_CLOEXEC), 0)
	if errno != 0 {
		return nil, errno
	}
	archivo := os.NewFile(descriptor, "runner-duplicado")
	var forma syscall.Stat_t
	if err := syscall.Fstat(int(descriptor), &forma); err != nil || uint64(forma.Dev) != r.dev || forma.Ino != r.ino || forma.Size != r.tamano || forma.Nlink != 0 || forma.Mode&0777 != 0600 {
		_ = archivo.Close()
		return nil, fmt.Errorf("duplicado del runner discrepante: err=%v", err)
	}
	return archivo, nil
}

func leerTraza(ruta string) (string, error) {
	datos, err := leerRutaAcotada(ruta, 65536)
	return string(datos), err
}

func (e *ejecucion) validarYRetirarEvidencia() (resultado error) {
	for _, ruta := range []string{e.salida, e.error} {
		datos, err := leerRutaAcotada(ruta, 65536)
		info, errStat := os.Stat(ruta)
		if err != nil || errStat != nil || len(datos) != 0 || !info.Mode().IsRegular() || info.Mode().Perm() != 0600 {
			resultado = errors.Join(resultado, fmt.Errorf("salida no vacía/acotada: bytes=%d err=%v/%v", len(datos), err, errStat))
		}
	}
	for _, ruta := range []string{e.traza, e.salida, e.error} {
		resultado = errors.Join(resultado, os.Remove(ruta))
	}
	return resultado
}

func leerRutaAcotada(ruta string, maximo int64) ([]byte, error) {
	fd, err := syscall.Open(ruta, syscall.O_RDONLY|syscall.O_CLOEXEC|syscall.O_NOFOLLOW, 0)
	if err != nil {
		return nil, err
	}
	archivo := os.NewFile(uintptr(fd), "entrada-acotada")
	defer archivo.Close()
	info, err := archivo.Stat()
	if err != nil || !info.Mode().IsRegular() || info.Size() < 0 || info.Size() > maximo {
		return nil, fmt.Errorf("entrada fuera de límite: bytes=%d err=%v", tamanoSeguro(info), err)
	}
	return leerArchivoAcotado(archivo, maximo)
}

func leerArchivoAcotado(archivo *os.File, maximo int64) ([]byte, error) {
	if _, err := archivo.Seek(0, 0); err != nil {
		return nil, err
	}
	datos, err := io.ReadAll(io.LimitReader(archivo, maximo+1))
	if err != nil {
		return nil, err
	}
	if int64(len(datos)) > maximo {
		return nil, errors.New("lectura excede el límite")
	}
	return datos, nil
}

func tamanoSeguro(info os.FileInfo) int64 {
	if info == nil {
		return -1
	}
	return info.Size()
}
