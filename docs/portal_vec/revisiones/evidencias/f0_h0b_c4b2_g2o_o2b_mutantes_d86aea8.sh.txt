comprobar_sha() {
    local archivo=$1 esperado=$2 etiqueta=$3
    local obtenido
    obtenido=$(sha256sum "$archivo" | awk '{print $1}')
    [[ "$obtenido" == "$esperado" ]] || {
        printf 'SHA %s discrepante: %s\n' "$etiqueta" "$obtenido" >&2
        exit 65
    }
}

limpiar() {

        estado=1
    fi
    if [[ -e "$temporal" ]]; then
        printf 'la limpieza temporal O2b dejó residuos\n' >&2
        estado=1
    fi
    exit "$estado"
}

reemplazar_unico() {
    local archivo=$1 anterior=$2 nuevo=$3
    perl -0777 - "$archivo" "$anterior" "$nuevo" <<'PERL'
use strict;
use warnings;
my ($ruta, $anterior, $nuevo) = @ARGV;
open my $entrada, '<:raw', $ruta or die "no se abre $ruta: $!\n";
local $/;
my $contenido = <$entrada>;
close $entrada or die "no se cierra $ruta: $!\n";
my $primero = index($contenido, $anterior);
die "patrón ausente en $ruta\n" if $primero < 0;
die "patrón múltiple en $ruta\n" if index($contenido, $anterior, $primero + length($anterior)) >= 0;
substr($contenido, $primero, length($anterior), $nuevo);
open my $salida, '>:raw', $ruta or die "no se escribe $ruta: $!\n";
print {$salida} $contenido or die "escritura incompleta $ruta: $!\n";
close $salida or die "no se cierra $ruta: $!\n";
PERL
}

mostrar_y_compilar() {
    local numero=$1
    local cambiado=g4 origen=$g4_origen
    if ((numero == 25)); then
        cambiado=g4
    fi
    printf '\n=== M%02d ===\n' "$numero"
    gofmt -w "$temporal/g1.go" "$temporal/g4.go" "$temporal/g5.go" "$temporal/g2.go" "$temporal/g3.go"
    [[ -z "$(gofmt -d "$temporal/g1.go" "$temporal/g4.go" "$temporal/g5.go" "$temporal/g2.go" "$temporal/g3.go")" ]] || {
        printf 'M%02d no queda en gofmt\n' "$numero" >&2
        return 1
    }
    diff -u --label G4_base --label "G4_M${numero}" "$origen" "$temporal/${cambiado}.go" || true
    printf 'SHA fuentes mutadas:\n'
    sha256sum "$temporal/g1.go" "$temporal/g4.go" "$temporal/g5.go" "$temporal/g2.go" "$temporal/g3.go"
    comprobar_sha "$temporal/g2.go" "$sha_g2" "G2 en M${numero}"
    comprobar_sha "$temporal/g3.go" "$sha_g3" "G3 en M${numero}"
    env GOENV=off GOTOOLCHAIN=go1.26.5 GOAMD64=v1 CGO_ENABLED=0 \
        "$go126" vet "$temporal/g1.go" "$temporal/g4.go" "$temporal/g5.go" "$temporal/g2.go" "$temporal/g3.go"
    env GOENV=off GOTOOLCHAIN=go1.26.5 GOAMD64=v1 CGO_ENABLED=0 \
        "$go126" build -a -trimpath -o "$temporal/supervisor" \
        "$temporal/g1.go" "$temporal/g4.go" "$temporal/g5.go" "$temporal/g2.go" "$temporal/g3.go"
}

oracle_ast() {
    local numero=$1
    local huella_g4
    huella_g4=$(sha256sum "$temporal/g4.go" | awk '{print $1}')
    preparar_ast_fuentes
    : >"$temporal/stdout"
    : >"$temporal/stderr"
    local estado=0
    ejecutar_ast --mutante "M${numero}" --sha-g4-mutante "$huella_g4" \
        >"$temporal/stdout" 2>"$temporal/stderr" || estado=$?
    [[ $estado -eq 1 && ! -s "$temporal/stdout" ]] || {
        printf 'M%02d SOBREVIVE_AST estado=%d\n' "$numero" "$estado" >&2
        cat "$temporal/stdout" "$temporal/stderr" >&2
        return 1
    }
    grep -q '^ast_o2b=fallo ' "$temporal/stderr" || {
        printf 'M%02d muerte AST sin diagnóstico O2b reconocido\n' "$numero" >&2
        cat "$temporal/stderr" >&2
        return 1
    }
    if grep -Eq 'huella discrepante|ledger de líneas discrepante' "$temporal/stderr"; then
        printf 'M%02d falso muerto por huella o ledger\n' "$numero" >&2
        cat "$temporal/stderr" >&2
        return 1
    fi
    printf 'M%02d MUERTO_AST estado=1 stderr=%q\n' "$numero" "$(cat "$temporal/stderr")"
}

comprobar_ast_conductual() {
    local numero=$1
    local huella_g4
    huella_g4=$(sha256sum "$temporal/g4.go" | awk '{print $1}')
    preparar_ast_fuentes
    : >"$temporal/stdout"
    : >"$temporal/stderr"
    local estado=0
    ejecutar_ast --mutante "M${numero}" --sha-g4-mutante "$huella_g4" \
        >"$temporal/stdout" 2>"$temporal/stderr" || estado=$?
    [[ $estado -eq 0 || $estado -eq 1 ]] || {
        printf 'M%02d control AST terminó en estado inesperado=%d\n' "$numero" "$estado" >&2
        return 1
    }
    if ((estado == 1)); then
        [[ ! -s "$temporal/stdout" ]] || return 1
        grep -q '^ast_o2b=fallo ' "$temporal/stderr" || return 1
        if grep -Eq 'huella discrepante|ledger de líneas discrepante' "$temporal/stderr"; then
            printf 'M%02d falso rechazo previo por huella o ledger\n' "$numero" >&2
            return 1
        fi
        printf 'M%02d CONTROL_AST_RECHAZA_SIN_CONTABILIZAR stderr=%q\n' "$numero" "$(cat "$temporal/stderr")"
        return 0
    fi
    [[ ! -s "$temporal/stderr" ]] || return 1
    printf 'M%02d CONTROL_AST_ACEPTA_MODO_MUTANTE\n' "$numero"
}

oracle_autoprueba() {
    local numero=$1 esperado=$2
    : >"$temporal/stdout"
    : >"$temporal/stderr"
    local estado=0
    "$temporal/supervisor" --autoprueba >"$temporal/stdout" 2>"$temporal/stderr" || estado=$?
    [[ $estado -eq 65 && ! -s "$temporal/stdout" && "$(cat "$temporal/stderr")" == "$esperado" ]] || {
        printf 'M%02d SOBREVIVE_O_MUERE_POR_ORACULO_AJENO estado=%d\n' "$numero" "$estado" >&2
        printf 'esperado=%q\nobtenido=%q\n' "$esperado" "$(cat "$temporal/stderr")" >&2
        return 1
    }
    printf 'M%02d MUERTO_AUTOPRUEBA estado=65 stderr=%q\n' "$numero" "$esperado"
}

esperado_autoprueba() {
    case "$1" in
        1) printf '%s' 'autoprueba del supervisor: control previo constructores: constructor confirmado discrepante' ;;
        2) printf '%s' 'autoprueba del supervisor: control previo constructores: receptor parcial no falló cerrado' ;;
        6|7|9) printf '%s' 'autoprueba del supervisor: control previo errores: secuencia inválida no se normalizó' ;;
        10|12|13) printf '%s' 'autoprueba del supervisor: control previo estados: secuencia de S2/S3 no se normalizó' ;;
        8) printf '%s' 'autoprueba del supervisor: control previo errores: secuencia inválida no se normalizó' ;;
        14) printf '%s' 'autoprueba del supervisor: control previo drenaje: causa o retirada sensible discrepante' ;;
        19) printf '%s' 'autoprueba del supervisor: control previo causas y EOF: causa o retirada sensible discrepante' ;;
        11) printf '%s' 'autoprueba del supervisor: control previo estados: estado vacío activo discrepante' ;;
        16) printf '%s' 'autoprueba del supervisor: control previo causas y EOF: causa o retirada sensible discrepante' ;;
        17) printf '%s' 'autoprueba del supervisor: control previo errores: causa o retirada sensible discrepante' ;;
        18) printf '%s' 'autoprueba del supervisor: control previo causas y EOF: ARMAR más EOF no canceló' ;;
        20) printf '%s' 'autoprueba del supervisor: control previo drenaje: drenaje coalescido discrepante' ;;
        21) printf '%s' 'autoprueba del supervisor: control previo drenaje: drenaje coalescido discrepante' ;;
        22) printf '%s' 'autoprueba del supervisor: control previo invariantes: invariante no retiró estado sensible' ;;
        26) printf '%s' 'autoprueba del supervisor: control previo tuplas O1b: tupla O1b imposible aceptada' ;;
        27|28|29) printf '%s' 'autoprueba del supervisor: control previo tuplas O1b: tupla O1b con lector incompatible aceptada' ;;
        30|31) printf '%s' 'autoprueba del supervisor: control previo tuplas O1b: tupla O1b con contador superior al límite aceptada' ;;
        *) return 1 ;;
    esac
}

# Línea base: formato, vet, build, AST y autoprueba deben ser positivos.
preparar
[[ -z "$(gofmt -d "$temporal/g1.go" "$temporal/g4.go" "$temporal/g5.go" "$temporal/g2.go" "$temporal/g3.go")" ]] || {
    printf 'la línea base no está en gofmt\n' >&2
    exit 65
}
env GOENV=off GOTOOLCHAIN=go1.26.5 GOAMD64=v1 CGO_ENABLED=0 \
    "$go126" vet "$temporal/g1.go" "$temporal/g4.go" "$temporal/g5.go" "$temporal/g2.go" "$temporal/g3.go"
env GOENV=off GOTOOLCHAIN=go1.26.5 GOAMD64=v1 CGO_ENABLED=0 \
    "$go126" build -a -trimpath -o "$temporal/supervisor" \
    "$temporal/g1.go" "$temporal/g4.go" "$temporal/g5.go" "$temporal/g2.go" "$temporal/g3.go"
preparar_ast_fuentes
ejecutar_ast
"$temporal/supervisor" --autoprueba
printf 'linea_base=OK sha_material=%s\n' "$sha_material"

muertos=0
seleccion=${VEP_MUTANTE_SOLO:-$(seq 1 31)}
for numero in $seleccion; do
    preparar
    aplicar_mutante "$numero"
    if ! mostrar_y_compilar "$numero"; then
        printf 'M%02d NO_CONTABILIZADO: formato/vet/build falló antes del oráculo\n' "$numero" >&2
        exit 1
    fi
    case "$numero" in
        3|4|5|15|23|24|25)
            oracle_ast "$numero"
            ;;
        *)
            comprobar_ast_conductual "$numero"
            oracle_autoprueba "$numero" "$(esperado_autoprueba "$numero")"
            ;;
    esac
    muertos=$((muertos + 1))
done

    [[ $muertos -eq 31 ]] || exit 1

else

    [[ $muertos -eq $esperados ]] || exit 1

fi
